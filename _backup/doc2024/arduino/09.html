7. データのスケーリング

数値にあまり差が無くわかりにくい場合や、値が大きすぎてSVGの中におさまりきらない場合は、比率は保ったまま範囲を変更する「スケーリング」を行うと良い。 

d3.json("./data.json")
.then((dataset) => {
  const svg =  d3.create("svg")
    .attr("viewBox", `0 0 ${width} ${height}`);

  /*
  domain()で変換前の範囲を、range()で変換後の範囲を指定する
  変換した範囲を変数「scale」に代入しておく
  */
  const scale = d3.scaleLinear()
    .domain([0, 90])
    .range([0, 400]);

  const g = svg.selectAll(".item")
    .data(dataset)
    .enter()
    .append("g")
    .classed("item", true)
    .attr("transform", (data, index) => `translate(0, ${index * 30})`);

  g.append("rect")
    .attr("x", 0)
    .attr("y", 0)
    .attr("width", (data) => scale(data.val)) /* 設定したscaleで値をキャストする */
    .attr("height", 20)
    .attr('fill', '#ff0000');

  g.append("text")
    .classed("label", true)
    .attr("font-size", "14")
    .attr("x", (data) => scale(data.val)) /* 設定したscaleで値をキャストする */
    .attr("y", 14)
    .text((data) => data.label);

  g.append("text")
    .classed("val", true)
    .attr("font-size", "14")
    .attr("x", 0)
    .attr("y", 14)
    .text((data) => data.val);

  g.append("image")
    .attr("href", (data) => data.url)
    .attr("width", 100);

  container.append(svg.node());
})
.catch((error) => {
});

7-1. 最大値と最小値を取得する

あらかじめ入力データの範囲がわかっていない場合は、d3.max(), d3.min()関数を使う。

d3.json("./data.json")
.then((dataset) => {
  const svg =  d3.create("svg")
    .attr("viewBox", `0 0 ${width} ${height}`);

  /*
  読み込んだJSONデータの値から、最大値（d3.max）と最小値（d3.min）を取得
  */
  const max = d3.max(dataset, (data) => data.val);
  const min = d3.min(dataset, (data) => data.val);
  console.log(max, min);

  const scale = d3.scaleLinear()
    .domain([0, max]) /* 変数に変更する */
    .range([0, 400]);

  const g = svg.selectAll(".item")
    .data(dataset)
    .enter()
    .append("g")
    .classed("item", true)
    .attr("transform", (data, index) => `translate(0, ${index * 30})`);

  g.append("rect")
    .attr("x", 0)
    .attr("y", 0)
    .attr("width", (data) => scale(data.val))
    .attr("height", 20)
    .attr('fill', '#ff0000');

  g.append("text")
    .classed("label", true)
    .attr("font-size", "14")
    .attr("x", (data) => scale(data.val))
    .attr("y", 14)
    .text((data) => data.label);

  g.append("text")
    .classed("val", true)
    .attr("font-size", "14")
    .attr("x", 0)
    .attr("y", 14)
    .text((data) => data.val);

  g.append("image")
    .attr("href", (data) => data.url)
    .attr("width", 100);

  container.append(svg.node());
})

8. アニメーション

g.append("rect")
  .attr("x", 0)
  .attr("y", 0)
  .attr("width", 0) /* 初期値に設定（設定前の記述は後で使用するのでコピペしておく） */
  .attr("height", 20)
  .attr('fill', '#ff0000')
  .transition() /* アニメーションの設定を開始する合図 */

  .ease(d3.easeLinear) /*イージングの種類 */
  .attr("width", (data) => scale(data.val)); /* アニメーション終了時の値 */

【やること】
  • 数字もアニメーションさせる。
  • グラフの色を変える。（JSONデータに16進数カラーコードを追加）

const width = 640;
const height = 480;
const imageW = 100;
const svg = d3.create("svg").attr("viewBox", `0 0 ${width} ${height}`);

d3.json("./data.json")
  .then((dataset) => {
    const max = d3.max(dataset, (data) => data.val);
    const min = d3.min(dataset, (data) => data.val);
    const scale = d3.scaleLinear().domain([0, max]).range([0, 400]);

    const g = svg
      .selectAll(".item")
      .data(dataset)
      .enter()
      .append("g")
      .classed("item", true)
      .attr("transform", (data, index) => `translate(20, ${index * 100 + 20})`);

    //要素で指定する場合
    g.append("rect")
      .attr("x", imageW + 20)
      .attr("y", 0)
      .attr("width", 0)
      .attr("height", 20)
      .attr("fill", (data) => data.color)
      .transition()
      .duration(1000)
      .delay((data, index) => index * 200)
      .ease(d3.easeBounceIn)
      .attr("width", (data) => scale(data.val));

    //クラス名で指定する場合
    g.append("text")
      .classed("label", true)
      .attr("font-size", "14")
      .attr("x", 0)
      .attr("y", 14)
      .attr("fill", (data) => data.color)
      .text((data) => data.label)
      .transition()
      .duration(1000)
      .delay((data, index) => index * 200)
      .ease(d3.easeBounceIn)
      .attr("x", (data) => scale(data.val) + imageW + 20);

    g.append("text")
      .classed("value", true)
      .attr("font-size", "0")
      .attr("x", imageW + 20)
      .attr("y", 14)
      .text((data) => data.val)
      .transition()
      .duration(1000)
      .delay((data, index) => index * 200)
      .ease(d3.easeBounceIn)
      .attr("font-size", "30");
