<!DOCTYPE html>
<html>
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>d3.js</title>
  <style>
    svg {
      background-color: #ddd;
    }
  </style>
</head>
<body>
  <div id="container"></div>

  <script src="https://cdn.jsdelivr.net/npm/d3@7"></script>
  <script>
   const width = 640;
  const height = 480;
  const imageW = 100;
  const svg = d3.create("svg").attr("viewBox", `0 0 ${width} ${height}`); //レスポンシブ対応

  d3.json("./data.json")
    .then((dataset) => {
      //データのスケーリング
      const max = d3.max(dataset, (data) => data.val);
      const min = d3.min(dataset, (data) => data.val);
      const scale = d3.scaleLinear().domain([0, max]).range([0, 400]);

      const g = svg
        .selectAll(".item")
        .data(dataset)
        .enter()
        .append("g")
        .classed("item", true)
        .attr("transform", (data, index) => `translate(20, ${index * 100 + 20})`); //グループは「transform」で操作

      //要素で指定する場合
      g.append("rect")
        .attr("x", imageW + 20)
        .attr("y", 35)
        .attr("width", 0) //値を初期値にする
        .attr("height", 20)
        .attr("fill", (data) => data.color)
        .transition() //アニメーションの設定
        .duration(1000)
        .delay((data, index) => index * 200)
        .ease(d3.easeBounceIn)
        .attr("width", (data) => scale(data.val));

      //クラス名で指定する場合
      g.append("text")
        .classed("label", true)
        .attr("font-size", 15)
        .attr("x", (data) => imageW + 20)
        .attr("y", 15)
        .attr("fill", (data) => data.color)
        .text((data) => data.label);

      g.append("text")
        .classed("value", true)
        .attr("font-size", 15)
        .attr("x", imageW + 20)
        .attr("y", 50)
        .text((data) => data.val)
        .transition()
        .duration(1000)
        .delay((data, index) => index * 200)
        .ease(d3.easeBounceIn)
        .attr("x", (data) => imageW + 20 + scale(data.val));

      g.append("image")
        .attr("href", (data) => data.url)
        .attr("width", imageW);

      //id名「container」の要素にsvgを追加
      container.append(svg.node());
  })
  .catch((error) => {});
  </script>
</body>
</html>

4. 簡潔に書く

4-1. 関数にする

【クリックイベントの中身を取り出し、関数に】
const onChange = () => {
  rect
    .transition()
    .duration(1000)
    .delay((data, index) => index * 200)
    .ease(d3.easeBounceIn)
    .attr("width", (data) => scale(data.val23));

  val
    .text((data) => data.val23)
    .transition()
    .duration(1000)
    .delay((data, index) => index * 200)
    .ease(d3.easeBounceIn)
    .attr("x", (data) => imageW + 20 + scale(data.val23));
};
【引数を追加し汎用性を持たせる】
const onChange = (key) => {
  rect
    .transition()
    .duration(1000)
    .delay((data, index) => index * 200)
    .ease(d3.easeBounceIn)
    .attr("width", (data) => scale(data[key])); //「data.val23」を「data[key]」に変更

  val
    .text((data) => data[key]) //「data.val23」を「data[key]」に変更
    .transition()
    .duration(1000)
    .delay((data, index) => index * 200)
    .ease(d3.easeBounceIn)
    .attr("x", (data) => imageW + 20 + scale(data[key])); //「data.val23」を「data[key]」に変更
};
【クリックで関数を実行】
const btn23 = document.querySelector("#btn23");
btn23.addEventListener("click", (e) => {
  onChange("val23");
});

const btn24 = document.querySelector("#btn24");
btn24.addEventListener("click", (e) => {
  onChange("val24");
});

4-2. カスタムデータ属性で制御

【forEachで回す】
const btn = document.querySelectorAll("button");

btn.forEach(elem => {
  elem.addEventListener('click', () => {
    onChange(elem.dataset.key);
  });
});
