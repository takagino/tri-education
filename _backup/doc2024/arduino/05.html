2-1. 図形の描画

【基本図形】
//点の描画：point( X座標, Y座標 );
point( 100, 200 );

//線の描画：line( X座標始点, Y座標始点, X座標終点, Y座標終点 );
line( 10, 10, 200, 180 );

//矩型の描画：rect( X座標, Y座標, 幅, 高さ );
rect( 10, 10, 200, 180 );

//楕円の描画：ellipse( X座標, Y座標, 幅, 高さ );
ellipse( 10, 10, 200, 180 );
【色の指定】
//背景色
background(0, 0, 0);

//グレースケール（白〜黒）の場合は値1個に省略できる
background(0);

//塗りの色
fill(255, 0, 0);

//塗りなし
noFill();

//線の色（16進数コードでもOK）
stroke("#0000ff");

//線なし
noStroke();
【透明度】
//値を1個追加することで、透明度を設定できる
background(0, 200);
fill(255, 0, 0, 100);

//16進数の書き方では無理っぽい
//stroke("#0000ff", 100);
【その他】
//カラーモードの変更：RGB、HSB、HSL
colorMode(HSL);

//矩型の基準点の変更
rectMode(CENTER);

//線の太さ・形状
strokeWeight(6);
strokeCap(ROUND);

4-2. 音声の解析

【analyze（周波数帯の割合？）】
let mySound;
let fft; //変数追加

function preload() {
  mySound = loadSound('groove.mp3');
}

function setup() {
  createCanvas(640, 480);
  colorMode(HSL);
  rectMode(CENTER);
  noStroke();
  mySound.amp(0.5);

  fft = new p5.FFT(); //fft(高速フーリエ変換)の作成
}

function draw() {

  if (mySound.isPlaying()) {
    let vol = map(mouseY, 0, height, 0, 1);
    vol = constrain(vol, 0, 1);
    mySound.amp(vol);

    //周波数帯の割合を取得
    //長さ「1024」の配列に値が格納される（0〜255）
    let analyze = fft.analyze();
    console.log(analyze);
  }
}

function keyPressed() {
  mySound.play();
}

function keyReleased() {
  mySound.stop();
}

【シンプルなサンプルコード】
let mySound;
let fft;

function preload() {
  mySound = loadSound('groove.mp3');
}

function setup() {
  createCanvas(640, 480);
  colorMode(HSL);
  rectMode(CENTER);
  angleMode(DEGREES);
  mySound.amp(0.5);
  fft = new p5.FFT();
}

function draw() {
  clear();

  if (mySound.isPlaying()) {
    let vol = map(mouseY, 0, height, 0, 1);
    vol = constrain(vol, 0, 1);
    mySound.amp(vol);

    let data = fft.analyze();
    console.log(data);

    for (let i = 0; i < data.length; i++) {
      let c = map(data[i], 0, 255, 0, 360);
      fill(c, 50, 50, 0.3);
      ellipse(320, 240, data[i], data[i]);
    }
  }
}

function keyPressed() {
  mySound.loop();
}

function keyReleased() {
  mySound.pause();
  clear();
}
