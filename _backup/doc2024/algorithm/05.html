function setup() {
  createCanvas(400, 400);
  noFill(); //塗りなし
}

function draw() {
  //3秒間クリックしなければ消す
  if (frameCount > 180) {
    clear();
  }

  //画面下は赤、上は青
  if (mouseY > 200) {
    stroke(255, 0, 0);
  } else {
    stroke(0, 0, 255);
  }

  //マウスをクリックしながら動かすと円を描画する
  //動かす速さに応じて円の大きさを変える
  if (mouseIsPressed) {
    frameCount = 0; //秒数を0にリセット
    let w = dist(mouseX, mouseY, pmouseX, pmouseY); //2点間の距離を計算する
    ellipse(mouseX, mouseY, w, w);
  }
}

3. 論理演算

論理演算とは、ある事象が真か偽かを判断する演算のこと。

3-1. 論理和（OR）

A または B
A B A || B
true true true
true false true
false true true
false false false
【例1】
function setup() {
  createCanvas(400, 400);
  noFill();
}

function draw() {
  if (mouseX < 100 || 300 < mouseX) {
    ellipse(mouseX, mouseY, 50, 50);
  }
}
☆練習
円が跳ね返るサンプルを、論理和を使ってまとめる。

3-2. 論理積（AND）

A かつ B
A B A && B
true true true
true false false
false true false
false false false
【例1】
function setup() {
  createCanvas(400, 400);
  noFill();
}

function draw() {
  if (mouseX > 200 && mouseY > 200) {
    ellipse(mouseX, mouseY, 50, 50);
  }
}
【例2】
function setup() {
  createCanvas(400, 400);
  noFill();
}

function draw() {
  if (mouseIsPressed && (mouseX > 200 || mouseY > 200)) {
    ellipse(mouseX, mouseY, 50, 50);
  }
}
☆練習
円が跳ね返るサンプルを改造し、マウスをクリックしている間のみ跳ね返るようにする。

3-3. 否定（NOT）

A ではない
A !A
true false
false true
【例1】
function setup() {
  createCanvas(400, 400);
  noFill();
}

function draw() {
  if (!mouseIsPressed) {
    ellipse(mouseX, mouseY, 50, 50);
  }
}
