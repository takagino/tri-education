Level.1

  1. 真ん中に円を描画する。
  2. クリックした場所に徐々に移動する。
let x;
let y;
let dx;
let dy;

function setup(){
  createCanvas(400, 400);
  colorMode(HSB);
  rectMode(CENTER);
  angleMode(DEGREES);

  x = width / 2;
  y = height / 2;
  dx = width / 2;
  dy = height / 2;
}

function draw(){
  clear();
  ellipse(dx, dy, 50, 50);
  dx += (x - dx) * 0.1;
  dy += (y - dy) * 0.1;
}

function mouseReleased(){
  x = mouseX;
  y = mouseY;
}
