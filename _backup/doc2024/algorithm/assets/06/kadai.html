【その1：kadai01.html】
  1. マウスを押している間だけ、軌跡に線を描く。
  2. マウスを動かす速さによって線の太さが変わり、3秒経つと消える。
  3. 線を描くたび、ランダムに色相が変わる。
let h = Math.random() * 360;
let count = 0;

function setup(){
  createCanvas(400, 400);
  rectMode(CENTER);
  colorMode(HSB);
}

function draw(){
  count++;
  stroke(h, 50, 100);

  let w = dist(mouseX, mouseY, pmouseX, pmouseY);
  strokeWeight(w);

  if (mouseIsPressed) {
    frameCount = 0;
    line(mouseX, mouseY, pmouseX, pmouseY);
  }

  if (frameCount > 180) {
    clear();
  }
}

function mouseReleased(){
  h = Math.random() * 360;
}
【その2：kadai02.html】
  1. 円が移動し、壁に当たったら跳ね返る。
  2. 跳ね返るたび、ランダムに色相が変わる。
let x = 50;
let y = 200;
let w = 50;
let speedX = 3;
let speedY = 2;

function setup(){
  createCanvas(400, 400);
  rectMode(CENTER);
  colorMode(HSB);
  noStroke();
}

function draw(){
  ellipse(x, y, w, w);
  x += speedX;
  y += speedY;

  if (x < w / 2 || x > width - w / 2) {
    speedX = -speedX;
    fill(Math.random() * 360, 50, 100);
  }

  if (y < w / 2 || y > height - w / 2) {
    speedY = -speedY;
    fill(Math.random() * 360, 50, 100);
  }
}

function mouseReleased(){
  clear();
}
【その3：kadai03.html】
  1. 位置、彩度、大きさがランダムな円を描く。
  2. マウスをクリックするたび、ランダムに色相が変わる。
let h = Math.random() * 360;

function setup(){
  createCanvas(400, 400);
  rectMode(CENTER);
  colorMode(HSB);
  noStroke();
}

function draw(){
  fill(h, Math.random() * 100, 100, 0.5);

  let w = Math.random() * 45 + 5;
  circle(Math.random() * width, Math.random() * height, w, w);
}

function mouseReleased(){
  h = Math.random() * 360;
}
