question02.html

  1. 横に移動する円のサンプルを改造し、縦（Y方向）にも動くようにする。 
const x = [];
const y = [];
const speedX = [];
const speedY = [];
const num = 10;

function setup() {
  createCanvas(400, 400);
  colorMode(HSB);
  rectMode(CENTER);

  for (let i = 0; i < num; i += 1) {
    x[i] = Math.random() * width;
    y[i] = Math.random() * height;
    speedX[i] = 3;
    speedY[i] = 2;
  }
}

function draw() {
  clear();

  for (let i = 0; i < num; i += 1) {
    ellipse(x[i], y[i], 20, 20);
    x[i] += speedX[i];
    y[i] += speedY[i];
    if (x[i] < 0 || x[i] > width) {
      speedX[i] = -speedX[i];
    }
    if (y[i] < 0 || y[i] > height) {
      speedY[i] = -speedY[i];
    }
  }
}
