Level.1

  1. クリックした場所に円が移動する。
let x;
let y;

function setup(){
  createCanvas(400, 400);
  colorMode(HSB);
  rectMode(CENTER);
  x = width / 2;
  y = width / 2;
}

function draw(){
  clear();
  ellipse(x, y, 50, 50);
}

function mouseReleased(){
  x = mouseX;
  y = mouseY;
}

Level.2 ※難問

  1. 徐々に移動する。
let x;
let y;
let dx;
let dy;

function setup(){
  createCanvas(400, 400);
  colorMode(HSB);
  rectMode(CENTER);
  dx = width / 2;
  dy = width / 2;
}

function draw(){
  clear();
  ellipse(dx, dy, 50, 50);
  dx += (x - dx) * 0.1;
  dy += (y - dy) * 0.1;
}

function mouseReleased(){
  x = mouseX;
  y = mouseY;
}
