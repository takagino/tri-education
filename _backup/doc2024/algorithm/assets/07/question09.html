question09.html

  1. ランダムな大きさ・色相の円または矩形を順番に描き、画面内に敷き詰める。
  2. 大きさは最小で5、最大で20とし、横縦ともに20個の図形を並べること。
  3. 画面をクリックすると最初から描画される。
let x = 0;
let y = 0;
let w = 20;

function setup(){
  createCanvas(400, 400);
  colorMode(HSB);
  rectMode(CENTER);
  noStroke();
}

function draw(){
  if (x < width) {
    fill(Math.random() * 360, 100, 100);

    let d = Math.random() * (w - 5) + 5;

    if (Math.random() < 0.5) {
      ellipse(x + w / 2, y + w / 2, d, d);
    } else {
      rect(x + w / 2, y + w / 2, d, d);
    }

    x += w;
  } else {
    if (y < height) {
      x = 0;
      y += w;
    }
  }
}

function mouseReleased(){
  clear();
  x = 0;
  y = 0;
}

question10.html

  1. 画面の左端中央から、上下にランダムに動く線を描く。
  2. 画面をクリックすると最初から描画される。
let x = 0;
let y;

function setup(){
  createCanvas(400, 400);
  colorMode(HSB);
  rectMode(CENTER);
  strokeWeight(6);
  y = height / 2;
}

function draw(){
  if (x < width) {
    let ry = Math.random() * 20 - 10;
    line(x, y, x += 1, y += ry);
  }
}

function mouseReleased(){
  clear();
  x = 0;
  y = height / 2;
}
