「アルゴリズム」繰り返し処理問題

【共通の条件】
  1. マウスの押すと指定の図形が描画される。
  2. マウスを離すと消える。

question01.html

  1. 画面のランダムな場所に、5本の線を描く。
  2. 線の太さは「10」、色はランダムとする。
function setup(){
  createCanvas(400, 400);
  colorMode(HSB);
  rectMode(CENTER);
  strokeWeight(10);
}

function draw(){
}

function mousePressed(){
  for(let i = 0; i < 5; i += 1){
    stroke(Math.random()*360, 100, 100);

    line(
      Math.random() * width, //1点目のx座標
      Math.random() * height, //1点目のy座標
      Math.random() * width, //2点目のx座標
      Math.random() * height //2点目のy座標
    );
  }
}

function mouseReleased(){
  clear();
}

question02.html

  1. 画面の高さ中央の1列に、8つの矩形を描く。
  2. 色はランダム、線は無しとする。
let w; //矩形の幅（まだ「width」は使用できないので宣言のみ）
let num = 8; //矩形を描く数

function setup(){
  createCanvas(400, 400);
  colorMode(HSB);
  rectMode(CENTER);
  noStroke();
  w = width / num; //画面の幅を矩形の数で割ることで、矩形1個の幅を求める
}

function draw(){
}

function mousePressed(){
  for (let i = 0; i < num; i += 1) { //矩形を描く数だけ繰り返す
    fill(Math.random() * 360, 100, 100);
    rect(w * i + w / 2, height / 2, w, w);
  }
}

function mouseReleased(){
  clear();
}

question03.html

  1. 画面の中心に8つの円を描く。
  2. 円の大きさは「50, 100, 150, 200, 250, 300, 350, 400」、色はランダム、線は無しとする。 
function setup(){
  createCanvas(400, 400);
  colorMode(HSB);
  rectMode(CENTER);
  noStroke();
}

function draw(){
}

function mousePressed(){
  for (let i = 8; i > 0; i -= 1) { //大きな円から順番に描いていく
    fill(Math.random() * 360, 100, 100);
    ellipse(width / 2, height / 2, 50 * i);
  }
}

function mouseReleased(){
  clear();
}
