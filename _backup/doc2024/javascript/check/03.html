Q1、Q2、Q3

ボタンをクリックしたらクラス名を取得し、p要素に表示するプログラムを作りたい。[ ]に入る適切なコードを答えなさい。

<button id="btn" class="hoge">click!</button>
<p id="result"></p>

<script>
  const btn = document.getElementById('btn');
  const result = document.getElementById('result');

  btn.addEventListener('click', function () {
    const val = [Q1].classList.[Q2]; //クラス名を取得する
    result.[Q3] = val; //p要素に表示する
  });
</script>
【正解】
<button id="btn" class="hoge">click!</button>
<p id="result"></p>

<script>
  const btn = document.getElementById('btn');
  const result = document.getElementById('result');

  btn.addEventListener('click', function () {
    const val = btn.classList.value;
    result.innerHTML = val;
  });
</script>

Q4、Q5、Q6

ボタンをクリックしたらbody要素にクラス名を追加し、背景色を黒色にするプログラムを作りたい。[ ]に入る適切なコードを答えなさい。

.color {
  background-color: #000;
}
<button id="btn">click!</button>

<script>
  const btn = document.getElementById('btn');

  btn.addEventListener('click', function () {
    document.[Q4].classList.[Q5]('[Q6]'); //body要素にクラス名を追加する
  });
</script>
【正解】
<button id="btn">click!</button>

<script>
  const btn = document.getElementById('btn');

  btn.addEventListener('click', function () {
    document.body.classList.add('color');
  });
</script>

Q7、Q8、Q9

ボタンをクリックするたびに、ボタンにクラス名「hoge」を追加したり削除したりを交互に繰り返すプログラムを作りたい。[ ]に入る適切なコードを答えなさい。

<button id="btn">click!</button>

<script>
  const btn = document.getElementById('btn');

  btn.addEventListener('click', function () {
    if (btn.classList.[Q7]('hoge')) {
      btn.classList.[Q8]('hoge');
    } else {
      btn.classList.[Q9]('hoge');
    }
  });
</script>
【正解】
<button id="btn">click!</button>

<script>
  const btn = document.getElementById('btn');

  btn.addEventListener('click', function () {
    if (btn.classList.contains('hoge')) {
      btn.classList.remove('hoge');
    } else {
      btn.classList.add('hoge');
    }
  });
</script>
