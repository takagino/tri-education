【正解】
<button id="btn">名作コピーを見る</button>

<script>
  const btn = document.getElementById('btn');

  btn.addEventListener('click', function () {
    const random = Math.floor(Math.random() * 5);
    let copy;

    switch (random) {
      case 0:
        copy = 'そうだ 京都、行こう。';
        break;
      case 1:
        copy = 'NO MUSIC, NO LIFE.';
        break;
      case 2:
        copy = 'インテル入ってる';
        break;
      case 3:
        copy = '愛は食卓にある。';
        break;
      case 4:
        copy = 'ココロも満タンに';
        break;
      default:
        console.log('どれも一致しません');
    }

    btn.innerHTML = copy;
  });
</script>
