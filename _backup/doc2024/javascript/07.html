1. はじめに

1-1. 今までの復習

  • JavaScriptがサイト上でやっているのは、あるタイミングで、ある要素に、何かしらの変化を加えることだけ。
  • JavaScript上達のコツは、とにかく「console.log()」で値を可視化・確認すること。 
/* 要素を指定して変数に代入 */
const submit = document.getElementById('submit');

/* 変数内に格納された要素をクリックしたら、function内の命令が実行される */
submit.addEventListener('click', function(){
  submit.style.color = '#ff0000';
});

/* 属性値の取得・変更 */
let value = submit.getAttribute('value'); //取得
submit.setAttribute('value', '変更する'); //変更

/* 内容の取得・変更 */
let text = submit.innerHTML;  //取得
submit.innerHTML = '変更しました！'; //変更

/* クラス名の操作 */
submit.classList.contains; //含むかどうか調べる（true, false）
submit.classList.add('hoge'); //追加
submit.classList.remove('piyo'); //削除
submit.classList.replace('fuga', 'hogera'); //置き換え
submit.classList.toggle('hoge'); //追加と削除を繰り返す

  //条件式Cに合致しないときに行う処理

if(条件式A){
  //条件式Aに合致したときに行う処理

  //条件式Bに合致しないときに行う処理

  //上記の条件式全てに合致しないときに行う処理

【条件分岐（switch文）】
switch(条件式){
  case 値1:
    //条件式の値が値1に一致する場合に実行される処理
    break;
  case 値2:
    //条件式の値が値2に一致する場合に実行される処理
    break;
  default:
    //上記のどの値にも一致しない場合に実行される処理
}

}

for(初期化処理; ループ継続の条件式; ループごとの最後に実行される式){

}

【ランダム】
/*「0.0000...」から「9.9999...」までのランダムな小数を取得する */
let num = Math.random() * 10;

/* 小数を切り捨てて、「0」から「9」までのランダムな整数を取得する */
num = Math.floor(Math.random() * 10);

/* 「1」をプラスすることで、「1」から「11」までのランダムな整数を取得する */
num = Math.floor(Math.random() * 10) + 1;
