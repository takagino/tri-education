2-2. アニメーションのコントロール

let ani;

function preload() {
  ani = loadAni('assets/spritesheet.png', { frameSize: [32, 32], frames: 18 });
}

function setup() {
  createCanvas(640, 480);
}

function draw() {
  clear();

  if (kb.presses('p')) ani.play();
  if (kb.presses('o')) ani.play(0);
  if (kb.presses('down')) ani.frameDelay--;
  if (kb.presses('up')) ani.frameDelay++;
  if (kb.presses('s')) ani.stop();
  if (kb.presses('r')) ani.rewind();
  if (kb.presses('l')) ani.loop();
  if (kb.presses('n')) ani.noLoop();
  if (kb.presses('left')) ani.previousFrame();
  if (kb.presses('right')) ani.nextFrame();
  if (kb.presses('5')) ani.frame = 5;
  if (kb.pressing('x')) ani.scale.x = -1;
  if (kb.pressing('y')) ani.scale.y = -1;
  if (kb.pressing('1')) ani.scale = 1;
  if (kb.pressing('2')) ani.scale = 2;

  animation(ani, width / 2, height / 2);
}

2-3. スプライトの作成

【スプライトとは？】
スプライトとは、一言で言うとゲームに登場する"キャラクター"の事。スプライトでアニメーションを作ることで、衝突判定や物理演算など様々な機能を利用することができる。

let ani, spr;

function preload() {
  ani = loadAni('assets/spritesheet.png', { frameSize: [32, 32], frames: 18 });
}

function setup() {
  createCanvas(640, 480);

  spr = new Sprite();
  spr.addAni('hiyoko', ani);
  spr.x = random(width);
  spr.y = random(height);
}

function draw() {
  clear();
}

【アニメーションを切り替える】
let ani01, ani02, spr;

function preload() {
  ani01 = loadAni('assets/spritesheet.png', { frameSize: [32, 32], frames: 18 });
  ani02 = loadAni('assets/bomb1.png', 4);
}

function setup() {
  createCanvas(640, 480);

  spr = new Sprite();
  spr.addAni('hiyoko', ani01);
  spr.addAni('bomb', ani02);
}

function draw() {
  clear();

  if (kb.presses('1')) spr.changeAni('hiyoko');
  if (kb.presses('2')) spr.changeAni('bomb');
}

3. Arduinoとの連携

3-1. Arduino側の設定（1回やればOK）

  1. PCとArduinoを接続。
  2. 「Arduino IDE」を立ち上げる。
  3. 『Select Board』の欄が「Arduino Uno」になっていることを確認。
  4. 『File > Examples > Firmata > StandardFirmataPlus』を開く。
  5. 検証 → 書き込み
  6. 書き込みが完了したらArduinoは終了する。

3-2. 距離センサーの回路

3-3. モジュールのインストール

  1. VSCodeでターミナルを表示。
  2. 以下のコードを入力し「Enter」。（初期設定） 
    npm init -y
  3. さらに、以下のコードを入力し「Enter」。（モジュールのインストール） 
    npm install socket.io johnny-five
  4. フォルダ内に「app.js」と「index.html」を新規作成。

  1. 「app.js」に以下のように記述。

3-6. センサーに反応させる

  1. 「app.js」を以下のように変更。 
    /* app.js */
const fs = require('fs');
const http = require('http');

const hostname = '127.0.0.1';
const port = 3000;

const index = fs.readFileSync('./index.html', 'utf-8');
const sheet = fs.readFileSync('./assets/spritesheet.png');

const server = http.createServer((req, res) => {
  const pathName = req.url;

  if (pathName === '/') {
    res.writeHead(200, {
      'Content-type': 'text/html'
    });
    res.end(index);
  } else if (pathName === '/sheet.png') {
    res.writeHead(200, {
      'Content-type': 'image/png'
    });
    res.end(sheet);
  } else {
    res.writeHead(404);
    res.end('Not Found...');
  }
});

/* 以下、変更 */
const five = require('johnny-five');
const board = new five.Board();

board.on("ready", () => {
  console.log("Arduino ready!");

  const sensor = new five.Sensor({
    pin: "A0",
    freq: 30
  });

  server.listen(port, hostname, () => {
    console.log(`Server running at http://${hostname}:${port}/`);

    const io = require("socket.io")(server);

    io.on("connection", (socket) => {
      console.log(`Connection completed!`);

      sensor.on("change", () => {
        io.emit('sensor_value', sensor.value);
      });

    });
  });
});
  2. 「index.html」のスクリプトを以下のように変更。 
    <script>
  let val = 0; //変数追加
  const socket = io();

  //センサーの値を代入
  socket.on('sensor_value', (recieve) => {
    val = recieve;
  });

  let ani, spr;

  function preload() {
    ani = loadAni('/sheet.png', { frameSize: [32, 32], frames: 18 });
  }

  function setup() {
    createCanvas(640, 480);

    spr = new Sprite();
    spr.addAni('hiyoko', ani);
  }

  function draw() {
    clear();

    //値の加工
    let min = 100;
    let max = 600;
    val = constrain(val, min, max);
    let frameNum = floor(map(val, min, max, 1, 17));
    console.log(frameNum);
  }
</script>

4. 課題

スプライトシートでアニメーションを作成し、距離センサーで動かす。

【例えば】
  • 距離センサーの値によってフレーム数を切り替える。
  • 距離センサーの値によってキャラクターを動かす。
  • 距離センサーの値が一定以上になったら、キャラクターが増殖する。（フレームレートを小さくすること「frameRate」）
