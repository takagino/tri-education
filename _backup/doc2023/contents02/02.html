1. 復習

下図のように回路を組み、LEDを光らせてみる。

  • ※「5V > 抵抗器 > LED > GND」へと流れる電流の道筋を意識すること。
【デジタル出力】
5Vに挿さっているジャンプワイヤをD12へ変更し、以下のプログラムを新規ファイルに書き込む。（※検証を忘れずに）
//LEDに接続したピンの番号を変数に代入
//JSとは違い、データ型を明記する必要がある（int：整数）
int ledPin = 12;

//setupは最初に1度だけ実行される
void setup(){
  //LEDに接続したピンのモードをOUTPUT(出力)に設定
  //デジタル信号は入出力を同じピンで行うため、指定が必要
  pinMode(ledPin, OUTPUT);
}

//loopはArduinoボードの電源がオンであるかぎり、繰り返し実行される
void loop(){
  //LEDに接続したピンの値をHIGHにセットしてLEDを点灯
  digitalWrite(ledPin, HIGH);
  //1000ms(1s)待つ
  delay(1000);
  //LEDに接続したピンの値をLOWにセットしてLEDを消灯
  digitalWrite(ledPin, LOW);
  //1000ms(1s)待つ
  delay(1000);
}

2. スイッチでの制御

【デジタル入力】
int pin = 2;

void setup(){
  //シリアル通信のデータ転送レートを指定する。単位はbps(ビット/秒)
  Serial.begin(9600);
  //ピンのモードをINPUT(入力)に設定
  pinMode(pin, INPUT);
}

void loop(){
  //ピンに入力された値を取得し変数に代入
  int value = digitalRead(pin);
  //シリアルモニタに表示する（println：値を表示して改行する）
  Serial.println(value);

  //負担を軽減するために処理を遅らせる
  delay(30);
}

2-2. サーボモータを動かす

※実際の回路は「オレンジ」が 9ピン、「赤」が5V、「茶色」が GND。

//ソースコードを読み込む
#include <Servo.h>

int pin = 2;

//変数追加
int servoPin = 9;
Servo myservo;

void setup(){
  Serial.begin(9600);
  pinMode(pin, INPUT);

  //ピンを設定
  myservo.attach(servoPin);
}

void loop(){
  int value = digitalRead(pin);

  //ピンの値に応じて処理をする（サーボモーターの角度は0〜180）
  if(value == 0){
    myservo.write(180);
  }else{
    myservo.write(0);
  }

  Serial.println(value);
  delay(30);
}

【値に変化があった場合のみ】
#include <Servo.h>

int pin = 2;
int servoPin = 9;
Servo myservo;

//値を一時的に保存するための変数
int prev_value = 0;

void setup(){
  Serial.begin(9600);
  pinMode(pin, INPUT);
  myservo.attach(servoPin);
}

void loop(){
  int value = digitalRead(pin);

  //現在の値と前の値が違っていたら
  if(value != prev_value){
    if(value == 0){
      myservo.write(180);
    }else{
      myservo.write(0);
    }
  }

  //値を更新
  prev_value = value;

  Serial.println(value);
  delay(30);
}

3. センサーでの制御

【距離センサに値を取得する（アナログ入力）】
int sensorPin = 0;

void setup() {
  Serial.begin(9600);

  //アナログ信号は入力と出力のピンが別れているため、ピンモードの設定は必要ない
}

void loop() {
  //0ピンから値を読み取り、変数に代入
  int value = analogRead(sensorPin);

  Serial.println(value);
}
【サーボモーターに送る適切な値に変換する】
int sensorPin = 0;

void setup() {
  Serial.begin(9600);
}

void loop() {
  int value = analogRead(0);

  //map(変換したい値, 変換前の最小値, 変換前の最大値, 変換後の最小値, 変換後の最大値);
  //事前に変換前の最小値と最大値を調べておくこと
  int angle = map(value, 50, 650, 0, 180);

  //「print」→ メッセージを表示、「println」→ メッセージを表示して改行
  Serial.print(value);
  Serial.print(":");
  Serial.println(angle);
}

3-1. ゆらぎの軽減

【分解能を下げる】
#include <Servo.h>

int sensorPin = 0;
int servoPin = 9;
Servo myservo;

void setup() {
  Serial.begin(9600);
  myservo.attach(servoPin);
}

void loop() {
  int value = analogRead(sensorPin);

  //10で割って小数点を切り捨てる
  value = floor(value / 10);

  //10倍して桁数を戻す
  value = value * 10;

  int angle = map(value, 50, 650, 0, 180);

  myservo.write(angle);
  delay(100); //変化がわかりやすいように遅延を伸ばす

  Serial.print(value);
  Serial.print(":");
  Serial.println(angle);
}

※細かいゆらぎはなくなったが、動きがぎこちなくなってしまうため、今回は不採用。

【平均値化する】
#include <Servo.h>

int sensorPin = 0;
int servoPin = 9;
Servo myservo;

void setup() {
  Serial.begin(9600);
  myservo.attach(servoPin);
}

void loop() {
  //毎回「0」に戻す
  int value = 0;

  //値を30回分合計する
  for(int i = 0; i < 30; i++){
    value += analogRead(sensorPin);
  }

  //合計した回数で割って平均値を出す
  value = value / 30;

  int angle = map(value, 50, 650, 0, 180);

  myservo.write(angle);
  delay(100);

  Serial.print(value);
  Serial.print(":");
  Serial.println(angle);
}

※かなりゆらぎは軽減したが、処理が重くなってしまうため、今回は不採用。

【指数平滑（RCフィルタ）】
#include <Servo.h>

int sensorPin = 0;
int servoPin = 9;
Servo myservo;

//毎回リセットされないように、loop関数外に宣言する
int value = 0;

//データ型「float（小数）」、値は「0〜1」の範囲
float filter = 0.1;

void setup() {
  Serial.begin(9600);
  myservo.attach(servoPin);
}

void loop() {
  //前回のvalの値9割と、新しく読み取ったA0ピンの値1割を足す
  value = (1 - filter) * value + analogRead(sensorPin) * filter;

  int angle = map(value, 50, 650, 0, 180);

  myservo.write(angle);
  delay(30); //遅延を小さくする

  Serial.print(value);
  Serial.print(":");
  Serial.println(angle);
}

※値の変化が「なだらか」になってしまうが、ゆらぎはほぼなくなるため今回はこちらを採用。

4. 振動モーターを動かす

int sensorPin = 0;
int motorPin = 9;

int value = 0;
float filter = 0.1;

void setup() {
  Serial.begin(9600);
}

void loop() {
  value = (1 - filter) * value + analogRead(sensorPin) * filter;

  int power = map(value, 50, 650, 0, 160);

  analogWrite(motorPin, power);
  delay(30);

  Serial.print(value);
  Serial.print(":");
  Serial.println(power);
}
