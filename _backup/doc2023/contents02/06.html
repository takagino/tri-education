4. 座標変換

//角度と拡大率のキーを追加
let obj = {
  x: 0,
  y: 0,
  c: 0,
  r: 0,
  s: 0,
};

function setup() {
  createCanvas(640, 480);
  rectMode(CENTER);
  colorMode(HSL);
  noStroke();
}

function draw() {
  // clear();

  obj.x = Math.floor(Math.random() * 640);
  obj.y = Math.floor(Math.random() * 480);
  obj.c = Math.floor(Math.random() * 100);

  //ランダムな値を追加
  obj.r = Math.floor(Math.random() * 360);
  obj.s = Math.random() * 2 + 0.5;

  //座標空間を保存
  push();

  //移動、回転、拡大縮小
  translate(obj.x, obj.y);
  rotate(obj.r);
  scale(obj.s);

  fill(180, 50, obj.c, 0.5);

  //x座標とy座標を修正
  rect(0, 0, 50, 50);

  //保存した状態に戻す
  pop();
}

6-1. 改造「お絵かきツール」

【マウスを動かす速さによって線の太さを変える】
  1. 線を設定する2つの点を結んだ距離を求める。（リファレンスの「Math - Calculation」）
  2. 求めた距離を線の太さに反映する。
function setup() {
  createCanvas(640, 480);
  colorMode(HSL);
  strokeWeight(6);
  strokeCap(ROUND);
}

function draw() {
  if (mouseIsPressed) {
    let d = dist(mouseX, mouseY, pmouseX, pmouseY);
    let w = map(d, 0, 100, 6, 20);
    strokeWeight(w);
    line(mouseX, mouseY, pmouseX, pmouseY);
  }
}

function keyReleased() {
  if (keyCode === 32) {
    clear();
  }
}

7. 課題「センサーで模様を描く」

【条件】
  • 「スイッチ」と「距離センサ」を使って回路を組む。
  • 「スイッチ」を押している間だけ模様（線や円、矩形などで構成）を描画し、話すと消える。
  • 「距離センサ」の値によって模様の何かしらを変化させる。
【回路】
【手順とサンプルコード】
  1. 課題用のフォルダを作成し、VSCodeで開く。
  2. ターミナルを表示、以下のコードを入力し「Enter」。 
    npm init -y
  3. ターミナルに以下のコードを入力し「Enter」。 
    npm install socket.io johnny-five
  4. フォルダ内に新規ファイル「app.js」を作成。 
    touch app.js
  5. 「app.js」に以下のコードを記述。 
    /* app.js */

const fs = require('fs');
const http = require('http');

const hostname = '127.0.0.1';
const port = 3000;
const index = fs.readFileSync('./index.html', 'utf-8');

const server = http.createServer((req, res) => {
  res.writeHead(200, {
    'Content-type': 'text/html'
  });
  res.end(index);
});

const five = require('johnny-five');
const board = new five.Board();

board.on("ready", () => {
  console.log("Arduino ready!");

  //ボタンの設定
  const button = new five.Button({
    pin: 2,
    isPullup: true
  });

  //センサーの設定
  let sensor_value = 0;
  const sensor = new five.Sensor({
    pin: "A0",
    freq: 30
  });

  server.listen(port, hostname, () => {
    console.log(`Server running at http://${hostname}:${port}/`);

    const io = require("socket.io")(server);

    io.on("connection", (socket) => {
      console.log(`Connection completed!`);

      //センサーの値を変数に代入
      sensor.on("change", () => {
        sensor_value = sensor.value;
      });

      //ボタンが押されている時の動作
      button.on("hold", function () {
        io.emit('button_value', {
          sensor: sensor_value,
          button: 'hold'
        });
      });

      //ボタンを離した時の動作
      button.on("release", function () {
        io.emit('button_value', {
          sensor: sensor_value,
          button: 'release'
        });
      });

    });
  });
});
  6. フォルダ内に新規ファイル「index.html」を作成。 
    touch index.html
  7. 「index.html」に以下のコードを記述。
