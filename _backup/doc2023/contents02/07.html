6. Arduinoとの連携

6-1. Arduino側の設定（1回やればOK）

  1. PCとArduinoを接続。
  2. 「Arduino IDE」を立ち上げる。
  3. 『Select Board』の欄が「Arduino Uno」になっていることを確認。
  4. 『File > Examples > Firmata > StandardFirmataPlus』を開く。
  5. 検証 → 書き込み
  6. 書き込みが完了したらArduinoは終了する。

6-2. 距離センサーの回路

6-3. モジュールのインストール

  1. 新規フォルダを作成し、VSCodeで開く。
  2. VSCodeでターミナルを表示。
  3. 以下のコードを入力し「Enter」。（初期設定） 
    npm init -y
  4. さらに、以下のコードを入力し「Enter」。（モジュールのインストール） 
    npm install socket.io johnny-five
  5. フォルダ内に「app.js」「index.html」を新規作成。

const fs = require('fs');
const http = require('http');

const hostname = '127.0.0.1';
const port = 3000;

const index = fs.readFileSync('./index.html', 'utf-8');
const audio = fs.readFileSync('./groove.mp3');

const server = http.createServer((req, res) => {
  const pathName = req.url;

  if (pathName === '/') {
    res.writeHead(200, {
      'Content-type': 'text/html'
    });
    res.end(index);
  } else if(pathName === '/groove.mp3') {
    res.writeHead(200, {
      'Content-type': 'audio/mpeg'
    });
    res.end(audio);
  } else {
    res.writeHead(404);
    res.end('Not Found...');
  }
});

const five = require('johnny-five');
const board = new five.Board();

/* 以下、変更 */
board.on("ready", () => {
  console.log("Arduino ready!");

  const sensor = new five.Sensor({
    pin: "A0",
    freq: 30
  });

  server.listen(port, hostname, () => {
    console.log(`Server running at http://${hostname}:${port}/`);

    const io = require("socket.io")(server);

    io.on("connection", (socket) => {
      console.log(`Connection completed!`);

      sensor.on("change", () => {
        io.emit('sensor_value', sensor.value);
      });

    });
  });
});
