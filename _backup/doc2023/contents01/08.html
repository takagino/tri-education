【Intersection Observer API の使い方】
  1. IntersectionObserverクラスからインスタンスを作成する。
  2. 監視される要素をターゲットにする。
  3. 実行される関数を設定する。
  4. オプションを設定する。
【インスタンスを作成する】
let observer = new IntersectionObserver();

/* IntersectionObserverには少なくとも1つの引数が必要なため、エラーが出る */
【監視される要素をターゲットにする】
let observer = new IntersectionObserver();

/* 追加 */
let targets = document.querySelectorAll('.box');
targets.forEach((target) => {
  observer.observe(target);
});
【実行される関数を設定する】
/* 関数を設定 */
let callback = (entries, observer) => {
  entries.forEach((entry) => {
    console.log(entry);
  });
};

/* 引数を追加 */
let observer = new IntersectionObserver(callback);

let targets = document.querySelectorAll('.box');
targets.forEach((target) => {
  observer.observe(target);
});
【コンソールに出す値を調整】
/* 取得できる値*/
entry.boundingClientRect
entry.intersectionRatio
entry.intersectionRect
entry.isIntersecting
entry.rootBounds
entry.target
entry.time

/* 変更 */
let callback = (entries, observer) => {
  entries.forEach((entry) => {
    console.log(entry.target.textContent, entry.isIntersecting);
  });
};

☆コンソールの値を見ながら動きを把握する。

【オプションを設定する】
/* オプションを設定 */
let options = {
  root: null,
  rootMargin: '0px',
  threshold: 0
}

let callback = (entries, observer) => {
  entries.forEach((entry) => {
    console.log(entry.target.textContent, entry.isIntersecting);
  });
};

/* 引数を追加 */
let observer = new IntersectionObserver(callback, options);

let targets = document.querySelectorAll('.box');
targets.forEach((target) => {
  observer.observe(target);
});


/*
オプション項目
root
ターゲットとなる要素が見えるかどうかを判定するためのベース部分を指定する。デフォルトはブラウザーのビューポート。

rootMargin
交差を検知する、上記 root からの距離。
記述方法はCSSの margin とほぼ同じだが、単位は px か % に限られる。デフォルトは0px。

threshold
関数を実行するタイミングを 0〜1 の間で記述する。ターゲットとなる要素が見え始めた瞬間と見え終わりの瞬間が 0、半分通過したときは 0.5、すべて見えている状態が 1。
[0, 0.5, 1] のように配列形式でも記述できる。デフォルトでは0。
*/
【画面に表示されたらクラス名を追加する】
let options = {
  root: null,
  rootMargin: '0px',
  threshold: 0
}

let callback = (entries, observer) => {
  entries.forEach((entry) => {
    /* 変更 */
    if (entry.isIntersecting) {
      entry.target.classList.add('in');
    } else {
      entry.target.classList.remove('in');
    }
  });
};

let observer = new IntersectionObserver(callback, options);

let targets = document.querySelectorAll('.box');
targets.forEach((target) => {
  observer.observe(target);
});

let targets = document.querySelectorAll('.box');

/* まずは1つだけ */
window.addEventListener('scroll', () => {
  const rect = target[1].getBoundingClientRect();

【要素が画面内に表示されているか判定する】
let targets = document.querySelectorAll('.box');

window.addEventListener('scroll', () => {
  const rect = targets[1].getBoundingClientRect();

  /* 変更 */
  const isIn = 0 <= rect.top && rect.bottom <= window.innerHeight;
  console.log(rect.top, rect.bottom, isIn);
});


/* 要素が一部でも画面内に表示されている場合 */
const isIn = 0 <= rect.bottom && rect.top <= window.innerHeight;
【画面に表示されたらクラス名を追加する】
let targets = document.querySelectorAll('.box');

window.addEventListener('scroll', () => {
  const rect = targets[1].getBoundingClientRect();
  const isIn = 0 <= rect.top && rect.bottom <= window.innerHeight;

  /* 追加 */
  if(isIn && !targets[1].classList.contains('in')){
    targets[1].classList.add('in');
  }else if(!isIn && targets[1].classList.contains('in')){
    targets[1].classList.remove('in');
  }
});
【全ての要素に適応する】
let targets = document.querySelectorAll('.box');

window.addEventListener('scroll', () => {
  targets.forEach((target) => {
    const rect = target.getBoundingClientRect();
    const isIn = 0 <= rect.top && rect.bottom <= window.innerHeight;

    if(isIn && !target.classList.contains('in')){
      target.classList.add('in');
    }else if(!isIn && target.classList.contains('in')){
      target.classList.remove('in');
    }
  });
});
【読み込み時やリサイズ時にも適応する】
let targets = document.querySelectorAll('.box');

/* 関数にまとめる */
const checkIn = (element) => {
  element.forEach((target) => {
    const rect = target.getBoundingClientRect();
    const isIn = 0 <= rect.top && rect.bottom <= window.innerHeight;

    if (isIn && !target.classList.contains('in')) {
      target.classList.add('in');
    } else if (!isIn && target.classList.contains('in')) {
      target.classList.remove('in');
    }
  });
};

/* イベント追加 */
window.addEventListener('load', () => {
  checkIn(targets);
});

window.addEventListener('resize', () => {
  checkIn(targets);
});

window.addEventListener('scroll', () => {
  checkIn(targets);
});

【画面に表示されてからのスクロール量を取得する】
let targets = document.querySelectorAll('p');

const checkIn = (element) => {
  element.forEach((target) => {
    const rect = target.getBoundingClientRect();
    const isIn = 0 <= rect.top && rect.bottom <= window.innerHeight;

    if (isIn && !target.classList.contains('in')) {
      target.classList.add('in');
    } else if (!isIn && target.classList.contains('in')) {
      target.classList.remove('in');
    }

    /* 追加 */
    if (isIn) {
      let move = window.innerHeight - rect.top;
      target.style.transform = `translateX(${move}px)`;
    }
  });
};

window.addEventListener('load', () => {
  checkIn(targets);
});

window.addEventListener('resize', () => {
  checkIn(targets);
});

window.addEventListener('scroll', () => {
  checkIn(targets);
});
【カスタムデータ属性で移動量を調整する】
<div class="container">
  <p data-rate="0.2">コードを書いていると、</p>
  <p data-rate="-0.2">食べることも忘れて</p>
  <p data-rate="0.5">ついつい熱中してしまう。</p>
</div>
let targets = document.querySelectorAll('p');

const checkIn = (element) => {
  element.forEach((target) => {
    const rect = target.getBoundingClientRect();
    const isIn = 0 <= rect.top && rect.bottom <= window.innerHeight;

    if (isIn && !target.classList.contains('in')) {
      target.classList.add('in');
    } else if (!isIn && target.classList.contains('in')) {
      target.classList.remove('in');
    }

    if (isIn) {
      /* 変更 */
      let move = (window.innerHeight - rect.top) * target.dataset.rate;
      target.style.transform = `translateX(${move}px)`;
    }
  });
};

window.addEventListener('load', () => {
  checkIn(targets);
});

window.addEventListener('resize', () => {
  checkIn(targets);
});

window.addEventListener('scroll', () => {
  checkIn(targets);
});
