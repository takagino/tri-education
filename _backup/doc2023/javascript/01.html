1. 復習

1-1. 基本事項

【要素の指定】
document.getElementsByTagName('要素名');
document.getElementById('id名');
document.getElementsByClassName('class名');
document.getElementsByName('name属性値');

document.querySelector('セレクタ'); //「.btn」など（cssセレクタと同じ書き方）
document.querySelectorAll('セレクタ');
【イベントの登録】
element.addEventListener('click', function(){
  element.style.color = '#ff0000';
});


/* アロー関数
--------------------*/
element.addEventListener('click', () => {
  element.style.color = '#ff0000';
});


/*「querySelectorAll」で要素を指定した場合、配列風オブジェクトになることに注意
--------------------*/
const btn = document.querySelectorAll('button');

for(let i = 0; i < btn.length; i++){
  btn[i].addEventListener('click', () => {
    btn[i].style.color = '#ff0000';
  });
}

1-2. 変数

【変数の宣言】
/* var：古い書き方（今は使わない）
--------------------*/
var val = "#000";


/* let：変数
--------------------*/
let num = 1;


/* const：値が変わらない変数（定数）
--------------------*/
const title = "JavaScript";
【演算子】
let x = 1;

x = x + 1;
console.log(x);


/* 自己代入演算子
--------------------*/
x -= 1;
console.log(x);


/* インクリメント演算子、デクリメント演算子
--------------------*/
x++;
console.log(x);


/* 前置と後置で結果が異なることに注意
--------------------*/
console.log(x--);
console.log(--x);


/* 割った時の余り
--------------------*/
console.log(x%5);
【その他計算】
/* 文字列同士の結合
--------------------*/
console.log("Java" + "Script");


/* 数字と文字列の計算
--------------------*/
console.log(0 + "1");


/* 変数と文字列の結合
--------------------*/
let title = "Java"
console.log(title + "Script");


/* テンプレートリテラル
--------------------*/
console.log(`${title}Script`);

1-3. 配列と繰り返し

【配列（Array）】
/* 配列の宣言
--------------------*/
const arr = ["あ","い","う","え","お"];


/* 要素の取得
--------------------*/
console.log(arr[0]);


/* 値を代入
--------------------*/
arr[0] = "か";


/* 空の配列を宣言
--------------------*/
let empty = [];


/* 配列の値を削除する場合は空の配列で上書きする
--------------------*/
empty = [];


/*
【注意】
「const」で宣言した配列は、値の上書きは可。配列自体の上書きはエラー。
*/
【for文】
for (初期値化式; 条件式; 更新式){
  forブロック //条件式に合致している間、繰り返し処理される
};

let arr = ["あ","い","う","え","お"];

for(let i = 0; i < arr.length; i++){
  console.log(arr[i]);
};
【while文】
while(条件式){
  whileブロック //条件式に合致している間、繰り返し処理される
};

let num = 0;

while(num < 5){
  console.log(num);
  num += 1;
};

1-4. 条件分岐

【if文】
if(条件式){
  条件式に合致したときに行う処理
};

if(条件式){
  条件式に合致したときに行う処理
}else{
  それ以外のときに行う処理
};

if(条件式1){
  条件式1に合致したときに行う処理
}else if(条件式2){
  条件式2に合致したときに行う処理
}else{
  それ以外のときに行う処理
};
【switch文】
switch(条件式){
  case 値1:
    //条件式の値が値1に一致する場合に実行される処理
    break;
  case 値2:
    //条件式の値が値2に一致する場合に実行される処理
    break;
  default:
    //上記のどの値にも一致しない場合に実行される処理
};

1-5. 関数とタイマー処理

【関数（Function）】
/* 関数の宣言と実行
--------------------*/
function testFunc(){
  console.log("関数実行");
}

testFunc();


/* 引数
--------------------*/
function testFunc(text){
  console.log(`${text}実行`);
}

testFunc("関数");


/* 戻り値
--------------------*/
function testFunc(val1, val2){
  return val1 + val2;
}

let num = testFunc(10, 20);
console.log(num);


/* 関数式
--------------------*/
const testFunc = function(val1, val2){
  return val1 + val2;
}


/*
【注意】
関数宣言と関数式の違いは、「関数名の重複」や「実行文の記述位置」でエラーがでるかどうか。
*/


/* アロー関数
--------------------*/
const testFunc = (val1, val2) => {
  return val1 + val2;
}
【setInterval】
//setInterval：一定時間ごとに特定の処理を繰り返す

let count = 0;
let timer; //変数を宣言しておく

const countUp = () => {
  console.log(count++);
  if(count > 5){
    clearInterval(timer); //処理を止める
  };
};

timer = setInterval(countUp, 1000);
【setTimeout】
//setTimeout：一定時間後に特定の処理をおこなう（繰り返さずに一度だけ）

let count = 0;
let timer;

const countUp = () => {
  console.log(count++);

  timer = setTimeout(countUp, 1000); //関数内にsetTimeoutを書くことで繰り返し処理となる
  if(count > 5){
    clearTimeout(timer); //処理を止める
  };
};

timer = setTimeout(countUp, 1000);

【Level.01】

見本

  • ボタンをクリックすると数が増える。

見本

見本

  • ボタン上でマウスを押すと画像が「on」になる。

見本

  • ストップボタンをクリックするとタイマーが止まる。
  • 再度スタートボタンをクリックすると、ゼロから再スタートする。

  • 100でストップすると「すごい！」と表示される。
【Level.03】

見本

  • スタートボタンをクリックするとタイマーがスタートし、数が増えていく。
  • ストップボタンをクリックするとタイマーが止まる。
  • 再度スタートボタンをクリックすると、ゼロから再スタートする。

見本

2-4. じゃんけんゲーム「janken」

【Level.01】

見本

  • 「グー、チョキ、パー」のボタンをクリックすると、自分の手がそれに応じた画像に変わる。
  • 同時に、相手の手がランダムな画像に変わる。
  • 同時に、じゃんけんの「勝ち・負け・あいこ」を判定し、結果のテキストに表示する。
  • 勝ちの場合は連勝記録が伸びる。
