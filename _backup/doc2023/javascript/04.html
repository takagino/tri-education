1-1. 文字列の操作

組み込みオブジェクト「String」

【基本操作】
/* 文字列の設定
--------------------*/
const str = "hello";


/* 文字列と数値を結合すると、数字は文字列に変換される
--------------------*/
console.log(10 + str);


/* 文字列の長さを取得
--------------------*/
console.log(str.length);


/* 1文字だけ取得する
--------------------*/
console.log(str[1]);


/* 部分的に取得する
--------------------*/
const strSub = str.substring(2, 4);
console.log(strSub);
【配列と文字列】
const str = "hello";

/* スプレッド演算子で配列に分配
--------------------*/
const strArr = [...str];
console.log(strArr);


/* 配列の値を結合し文字列を生成する（join）
--------------------*/
const joinArr = strArr.join('/');
console.log(joinArr);


/* 文字列の値を分割し配列を生成する（split）
--------------------*/
const splitArr = joinArr.split('/');
console.log(splitArr);


/* 分割代入への応用
--------------------*/
const [hour, minute, second, ...other] = joinArr.split('/');
console.log(hour);
console.log(other);
【練習】
  1. 自分の漢字の名前の1文字1文字の間に「・」を追加し、コンソールに表示する。 
    const name = "高木寛貴";
// 「高・木・寛・貴」と表示されるように
const name = "高木寛貴";

const nameArr = [...name]; //["高","木","寛","貴"]

const nameJoin = nameArr.join('・'); //"高・木・寛・貴"
【様々なメソッド】
const str = "Webデザイン学科";

/* 切り出し（slice）
--------------------*/
console.log(str.slice(0, 3));


/* 置換・削除（replace）
--------------------*/
console.log(str.replace('Web', 'グラフィック'));


/* 検索（indexOf）
--------------------*/
console.log(str.indexOf('Web')); //文字列を含む場合：何文字目にあるか
console.log(str.indexOf('CG')); //文字列を含まない場合：-1


/* コピーして結合（repeat）
--------------------*/
console.log(str.repeat(3));


/* 文字列の延長（padStart、padEnd）
--------------------*/
console.log(str.padStart(20, '!'));
console.log(str.padEnd(20, '!'));

1-2. 数値の操作

【基本操作】
const scores = [10, 3, 9];
let sum = 0;

/* 合計値を求める
--------------------*/
for (const score of scores) {
  sum += score;
};


/* 平均値を求める
--------------------*/
const avg = sum / scores.length;

console.log(`${sum} : ${avg}`);
【Mathオブジェクト】
組み込みオブジェクト「Math」
/* 小数点以下切り捨て
--------------------*/
console.log(Math.floor(avg)); //7


/* 小数点以下切り上げ
--------------------*/
console.log(Math.ceil(avg)); //8


/* 四捨五入
--------------------*/
console.log(Math.round(avg)); //7


/* 指定の桁数で表示
--------------------*/
console.log(avg.toFixed(3)); //7.333
【練習】
  1. 1〜10までの数字が入った配列を作る。
  2. 配列に入れた数字を全部足す。
  3. それの平均値を出して、小数点第2位まで表示する。
  4. その値を2乗して、小数点で切り捨てる。
const arr = [];
for (let i = 1; i <= 10; i++) {
    arr.push(i);
}

let sum = 0;
for (const val of arr) {
  sum += val;
}; //55

const avg = sum / arr.length; //5.5

const newNum = avg.toFixed(2); //5.50

const numPow = Math.pow(newNum, 2); //30.25

const numFloor = Math.floor(numPow) //30

1-3. 日時の操作

組み込みオブジェクト「Date」

【基本操作】
/* Dateオブジェクトの取得
--------------------*/
const d = new Date();
console.log(d);


/* 個別の数値で取得
--------------------*/
console.log(`${d.getMonth() + 1} 月 ${d.getDate()} 日`);


/* UTC (協定世界時) での「1970年1月1日0時0分0秒」から現在までの経過時間をミリ秒単位で返す
--------------------*/
console.log(Date.now());
【特定の日時を取得する】
const d = new Date(1981, 0); // 1981/01/01 00:00:00
console.log(d);

d.setHours(10, 20, 30); // 時間を変更する（1981/01/01 10:20:30）

d.setDate(32); // 日時を変更する（1981/02/01 10:20:30）※範囲を超えると自動的に計算される

d.setDate(d.getDate() + 3); // 今の日時の何日後（1981/02/04 10:20:30）
【経過時間を取得する】
let nowTime;
let previousTime = Date.now();

window.addEventListener('click', () => {
  nowTime = Date.now();
  console.log(nowTime - previousTime); //秒数で表示する場合は1000で割る
  previousTime = nowTime;
});

2-3. タイピングの正誤を判定する

<script>
const target = document.querySelector('#target');
const word = "red"; //正解の文字列
let loc = 0; //何文字目を入力しているかを判別する変数（location）

document.addEventListener('keydown', e => {
  if (e.key === word[loc]) {
    loc++;

    // 1: _ed
    // 2: __d
    // 3: ___
    target.textContent = '_'.repeat(loc) + word.substring(loc);
  }
});
</script>

2-4. 早期リターンに変更する

<script>
const target = document.querySelector('#target');
const word = "red";
let loc = 0;

document.addEventListener('keydown', e => {
  /*
  メインとなる処理以外の例外ケースを除外し、メインの処理をわかりやすくすることを早期リターン、もしくはアーリーリターンと呼ぶ。
  */
  if (e.key !== word[loc]) {
    return
  }

  loc++;
  target.textContent = '_'.repeat(loc) + word.substring(loc);
});
</script>

word = words[Math.floor(Math.random() * words.length)];

2-6. 入力が完了したら次の単語を表示する

<script>
const target = document.querySelector('#target');
const words = ["red", "blue", "pink"];
let word;
let loc = 0;

//関数に変更する
const setWord = () => {
  word = words[Math.floor(Math.random() * words.length)];
  target.textContent = word;
  loc = 0;
}

//関数を実行する
setWord();

document.addEventListener('keydown', e => {
  if (e.key !== word[loc]) {
    return
  }

  loc++;
  target.textContent = '_'.repeat(loc) + word.substring(loc);

  //正解文字数と単語の文字数が一緒だったら
  if (loc === word.length) {
    setWord();
  }
});
</script>

2-7. 単語の重複を防ぐ

<script>
const target = document.querySelector('#target');
const words = ["red", "blue", "pink"];
let word;
let loc = 0;

const setWord = () => {
  //spliceメソッドで配列から削除する
  word = words.splice(Math.floor(Math.random() * words.length), 1)[0];
  target.textContent = word;
  loc = 0;
}

setWord();

document.addEventListener('keydown', e => {
  if (e.key !== word[loc]) {
    return
  }

  loc++;
  target.textContent = '_'.repeat(loc) + word.substring(loc);

  if (loc === word.length) {
    //全ての単語を入力した際の処理
    if (words.length === 0) {
      const result = document.querySelector('#result');
      result.textContent = 'Finished!';
      return;
    }

    setWord();
  }
});
</script>

2-8. ゲームをスタートさせる

<p id="target">Click to start!</p>
<p id="result"></p>

<script>
const target = document.querySelector('#target');
const words = ["red", "blue", "pink"];
let word;
let loc = 0;
let startTime; //ゲームスタートした時の時刻を記録するための変数を追加

const setWord = () => {
  word = words.splice(Math.floor(Math.random() * words.length), 1)[0];
  target.textContent = word;
  loc = 0;
}

//クリックイベントを追加
document.addEventListener('click', () => {
  startTime = Date.now();
  setWord();
});

document.addEventListener('keydown', e => {
  if (e.key !== word[loc]) {
    return
  }

  loc++;
  target.textContent = '_'.repeat(loc) + word.substring(loc);

  if (loc === word.length) {
    if (words.length === 0) {
      const result = document.querySelector('#result');
      const elapsedTime = ((Date.now() - startTime) / 1000).toFixed(2); //終了までにかかった時間を計算する
      result.textContent = `Finished! ${elapsedTime} seconds!`; //結果を表示
      return;
    }

    setWord();
  }
});
</script>

2-9. ゲーム中はクリックできないようにして完成

<script>
const target = document.querySelector('#target');
const words = ["red", "blue", "pink"];
let word;
let loc = 0;
let startTime;
let isPlaying = false; //ゲーム中かどうかを判別するための変数を追加

const setWord = () => {
  word = words.splice(Math.floor(Math.random() * words.length), 1)[0];
  target.textContent = word;
  loc = 0;
}

document.addEventListener('click', () => {
  //isPlayingが「true」の場合は何もしない
  if (isPlaying === true) {
    return;
  }

  isPlaying = true; //isPlayingが「false」の場合は「true」にする
  startTime = Date.now();
  setWord();
});

document.addEventListener('keydown', e => {
  if (e.key !== word[loc]) {
    return
  }

  loc++;
  target.textContent = '_'.repeat(loc) + word.substring(loc);

  if (loc === word.length) {
    if (words.length === 0) {
      const result = document.querySelector('#result');
      const elapsedTime = ((Date.now() - startTime) / 1000).toFixed(2);
      result.textContent = `Finished! ${elapsedTime} seconds!`;
      return;
    }

    setWord();
  }
});
</script>

4. 実践「ストップウォッチ」

見本

【実装する機能】
  1. 画面表示時は「スタートボタン」しかクリックできない。
  2. 「スタートボタン」をクリックしたらタイムが動き出す。
  3. タイムが動き出したら「ストップボタン」しかクリックできない。
  4. 「ストップボタン」をクリックしたらタイムが一時停止される。
  5. 一時停止時に再度「スタートボタン」をクリックするとタイムが動き出す。
  6. 一時停止時に「リセットボタン」をクリックするとタイムがリセットされる。
