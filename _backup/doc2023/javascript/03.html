1-1. オブジェクトとは

前回勉強した配列は複数の値に順番を付けてまとめたものでしたが、値に名前を付けて管理することができるオブジェクトを使うと、より便利になる。

/* 「x座標とy座標を格納するための配列」の場合
--------------------*/
const point = [100, 180];


/* オブジェクトの形でわかりやすくする
--------------------*/
const point = {x: 100, y: 180};


/* 改行を入れて見やすくする
--------------------*/
const point = {
  x: 100,
  y: 180,
};

console.log(point);

【オブジェクトの操作】
const point = {
  x: 100,
  y: 180,
};


/* 値の取得
--------------------*/
console.log(point.x); //ドット記法
console.log(point['y']); //ブラケット記法


/* 値の変更
--------------------*/
point.x = 120;
point['y'] = 200;


/* 値の追加
--------------------*/
point.z = 90;


/*値の削除
--------------------*/
delete point.y;

console.log(point);

2. オブジェクトに関する追加知識

2-1. スプレッド演算子と分割代入

/* スプレッド演算子
--------------------*/
const otherProps = {
  r: 4,
  color: 'red',
}

const point = {
  x: 100,
  y: 180,
  ...otherProps,
};

console.log(point);


/* 分割代入（プロパティ名と同じ変数に代入）
--------------------*/
const {x, color, ...others} = point;

console.log(x);
console.log(color);
console.log(others);

2-2. オブジェクトの値を全て取り出す

const point = {
  x: 100,
  y: 180,
  r: 4
};

/* 名前（キー）を取り出す
--------------------*/
const obj = Object.keys(point);

for(const key of obj){
  console.log(`キー: ${key}`);
};


/* 値を取り出す
--------------------*/
const obj = Object.values(point);

for(const value of obj){
  console.log(`値: ${value}`);
};


/* 名前（キー）と値を取り出す
--------------------*/
const obj = Object.entries(point);

for(const [key, value] of obj){
  console.log(`${key} : ${value}`);
};

2-3. 配列とオブジェクトの合わせ技

const teacher = [
  {
    name: "高木",
    age: 42,
    blood: "B"
  },
  {
    name: "田中",
    age: 35,
    blood: "AB"
  }
];

console.log(teacher[1].name);


/* 値を取り出す
--------------------*/
for (const member of teacher) {
  const obj = Object.entries(member);
  for (const [key, value] of obj) {
    console.log(`${key}: ${value}`);
  }
}

  const setMonster = () => {

    currentHp = monsters[currentMonster].hp;

  setMonster();

    currentHp--;

      currentMonster++;

      //関数を実行
      setMonster();
    }
  });
</script>

3-6. すべてのモンスターを倒したらクリア

<script>
  const monsters = [
    {
      name: "スライム",
      pic: "https://trident.takagino.net/doc2023/javascript/assets/03/click/images/slime.png",
      hp: 4
    },
    {
      name: "目玉",
      pic: "https://trident.takagino.net/doc2023/javascript/assets/03/click/images/eye.png",
      hp: 6
    },
    {
      name: "半魚人",
      pic: "https://trident.takagino.net/doc2023/javascript/assets/03/click/images/fish.png",
      hp: 8
    },
  ];

  const pic = document.querySelector('.pic');
  const name = document.querySelector('.name');
  const hp = document.querySelector('.hp');

  let currentHp;
  let currentMonster = 0;

  const setMonster = () => {
    pic.src = monsters[currentMonster].pic;
    name.textContent = monsters[currentMonster].name;
    hp.textContent = `HP:${monsters[currentMonster].hp}`;
    currentHp = monsters[currentMonster].hp;
  };

  setMonster();

  pic.addEventListener('click', () => {
    currentHp--;
    hp.textContent = `HP:${currentHp}`;

    if (currentHp <= 0) {
      currentMonster++;

      //if文に修正
      if (currentMonster >= monsters.length) {
        hp.textContent = `クリア！`;
      } else {
        setMonster();
      }
    }
  });
</script>
